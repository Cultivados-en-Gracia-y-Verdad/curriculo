% Anchorblock environment – handles H4 + H5 together
\newenvironment{anchorblock}[1]{%
  \par\vspace{-0.8em}

  % Anchor word
  {\ScriptureFont\itshape\color{\AnchorColor}\normalsize %
   \hspace*{2em}«#1»\par}

  \vspace{0.1em}

  % Begin indented content (SAFE)
  \begingroup
  \setlength{\leftskip}{4em}
  \setlength{\parindent}{0pt}
  \setlength{\parskip}{0pt}
  \color{\AnchorColor}
}{%
  \par
  \endgroup
  \vspace{0.8em}
}
