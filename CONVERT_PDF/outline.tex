\newenvironment{anchorblock}[1]{%

  \vspace{0.1em}

  % Begin indented content (SAFE)

  \setlength{\leftskip}{4em}
  \setlength{\parindent}{0pt}
  \setlength{\parskip}{0pt}
  \color{\AnchorColor}
}{%
  \par
  \endgroup
  \vspace{0.7em}
}
